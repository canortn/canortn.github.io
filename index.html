SwiftUI Clock v1

First version of my interactive SwiftUI clock — tap to advance time.

import SwiftUI

struct ContentView: View {
    @State var hour = 13
    @State var minuteSeg = 0

    var hourAngle: Double {
        let h = hour % 12
        let base = Double(h) * 30
        let step = Double(minuteSeg) * 7.5
        return (h == 0 && minuteSeg == 0) ? 0 : base + step
    }

    var body: some View {
        VStack(spacing: 16) {
            Text("Time: \(hour)")
            ZStack {
                if hour < 12 {
                    Circle().fill(.yellow).opacity(0.8).overlay(Circle().stroke(.white,lineWidth:4)).frame(width:200,height:200)
                } else {
                    Circle().fill(.blue).opacity(0.8).overlay(Circle().stroke(.white,lineWidth:4)).frame(width:200,height:200)
                }
                // minute hand + hour hand...
                Rectangle().fill(.white).frame(width:8,height:40).offset(x:0,y:-25).rotationEffect(.degrees(hourAngle))
                Circle().fill(.white).frame(width:10,height:10)
            }
            .onTapGesture {
                minuteSeg += 1
                if minuteSeg == 4 { minuteSeg = 0; hour += 1 }
                if hour == 24 { hour = 0 }
            }
        }.padding()
    }
}

import SwiftUI

struct ContentView: View {
    @State var id = 0
    let maxID = 15

    private var bits: [String] {
        Array(decimalToBinary(id)).map { String($0) }
    }

    func decimalToBinary(_ number: Int) -> String {
        let powers = [8,4,2,1]; var n = number%16; var s = ""
        for p in powers { if n >= p { s += "1"; n -= p } else { s += "0" } }
        return s
    }

    var body: some View {
        ZStack {
            Color.black.ignoresSafeArea()
            VStack(spacing:16) {
                Text("4-Bit ID Counter").font(.title).foregroundColor(.white).bold()
                HStack {
                    Text("Decimal: \(id)").foregroundColor(.white)
                    Text("Binary: \(decimalToBinary(id))").foregroundColor(.white)
                }.font(.title3)
                HStack(spacing:12) {
                    ForEach(0..<4, id:\.self) { i in
                        RoundedRectangle(cornerRadius:14)
                            .fill(bits[i]=="1" ? Color.green : Color.gray.opacity(0.6))
                            .frame(width:68,height:68)
                            .overlay(Text(bits[i]).font(.title).bold().foregroundColor(.white))
                    }
                }
                HStack(spacing:24) {
                    Button("Enroll (+1)") { id = id==maxID ? 0 : id+1 }
                    Button("Reset (0)") { id = 0 }
                }
            }.padding()
        }
    }
}

struct Triangle: Shape {
    func path(in rect: CGRect) -> Path {
        var path = Path()
        path.move(to: CGPoint(x:rect.midX, y:rect.maxY))
        path.addLine(to: CGPoint(x:rect.minX, y:rect.minY))
        path.addLine(to: CGPoint(x:rect.maxX, y:rect.minY))
        path.closeSubpath(); return path
    }
}

struct ContentView: View {
    @State private var scoop1: Color? = nil
    @State private var scoop2: Color? = nil
    @State private var scoop3: Color? = nil
    @State private var scoop4: Color? = nil
    @State private var scoop5: Color? = nil
    @State private var count = 0

    func colorFor(_ i: Int) -> Color {
        [Color.pink,.yellow,.mint,.purple,.blue][(i-1) % 5]
    }

    func add() {
        if scoop1==nil { scoop1=colorFor(1); count=1 }
        else if scoop2==nil { scoop2=colorFor(2); count=2 }
        else if scoop3==nil { scoop3=colorFor(3); count=3 }
        else if scoop4==nil { scoop4=colorFor(4); count=4 }
        else if scoop5==nil { scoop5=colorFor(5); count=5 }
        else { reset() }
    }

    func reset() { count=0; scoop1=nil; scoop2=nil; scoop3=nil; scoop4=nil; scoop5=nil }

    // body: ZStack with Triangle (cone) + Circle scoops stacked by offset

SwiftUI Candy Shop Simulation

Console-style sim tracking candy stock, prices, and best sellers over 60 minutes.

import SwiftUI

struct ContentView: View {
    @State private var report = ""

    var body: some View {
        ScrollView {
            Text(report.isEmpty ? "Running..." : report)
                .font(.system(.body, design:.monospaced)).padding()
        }
        .onAppear { report = simulate() }
    }

    func simulate() -> String {
        var chocolateStock=12; var gummyStock=10; var lollipopStock=12; var caramelStock=8
        var soldChocolate=0; var soldGummy=0; var soldLollipop=0; var soldCaramel=0; var cash=0
        var lines: [String] = []

        for minute in 1...60 {
            if minute%2==0 && gummyStock>0 { gummyStock-=1; soldGummy+=1; cash+=2; lines.append("m\(minute): Gummy sold") }
            if minute%3==0 && lollipopStock>0 { lollipopStock-=1; soldLollipop+=1; cash+=2; lines.append("m\(minute): Lollipop sold") }
            if minute%5==0 {
                if chocolateStock>0 { chocolateStock-=1; soldChocolate+=1; cash+=3; lines.append("m\(minute): Chocolate sold") }
                else if caramelStock>0 { caramelStock-=1; soldCaramel+=1; cash+=3 }
            }
            if minute%20==0 { chocolateStock+=2; gummyStock+=2; lollipopStock+=2; caramelStock+=2 }
        }

        lines.append("FINAL: Cash=$\(cash) | Best: Gummy(\(soldGummy))")
        return lines.joined(separator:"\n")
    }
}

SwiftUI Hot Chocolate Machine

import SwiftUI

func hotChocolateMachine() -> [String] {
    let startTemp = 20; var currentTemp = startTemp; var out: [String] = []
    for minute in 1...20 {
        currentTemp += 5
        out.append("Minute \(minute): \(currentTemp)°C")
        if currentTemp < 50      { out.append("→ Still cold.") }
        else if currentTemp <= 70 { out.append("→ Ready to serve!") }
        else if currentTemp <= 80 { out.append("→ Too hot! Wait.") }
        else { out.append("⚠ Auto-stop!"); break }
    }
    out.append("Total change: \(currentTemp-startTemp)°C")
    return out
}

struct GameLogic {
    let choices = ["✊","✋","✌️"]

    enum Outcome {
        case win, lose, tie
        var text: String {
            switch self { case .win: return "Kazandın!"; case .lose: return "Kaybettin!"; case .tie: return "Berabere!" }
        }
    }

    func randomComputerChoice() -> String { choices.randomElement() ?? "✊" }

    func determineWinner(userChoice:String,computerChoice:String)->Outcome {
        if userChoice==computerChoice { return .tie }
        if (userChoice=="✊"&&computerChoice=="✌️")||(userChoice=="✋"&&computerChoice=="✊")||(userChoice=="✌️"&&computerChoice=="✋") { return .win }
        return .lose
    }
}

Big Ideas

AP CSP — 5 Big Ideas, key concepts & reflections

01 Big Idea 1 — Creative Development

In this unit I learned how to turn an idea into a working program by planning, testing, presenting, and explaining my design. I documented everything with notes and screenshots, and made a short narration to show how the program works and why I built it.

1.1 — Collaboration

Good programs are rarely built alone. Collaboration means sharing tasks, reviewing each other's code, and combining different perspectives to produce better results. I worked with classmates on the Tic-Tac-Toe project — we split debugging tasks and caught errors faster together than I would have alone.

  • Key concept: Computing is a collaborative discipline. Pair programming, code reviews, and feedback loops all lead to stronger programs.
  • Real-world example: Open-source projects on GitHub rely entirely on contributors collaborating without ever meeting in person.
  • Reflection: I learned that explaining my code to someone else often reveals bugs I missed. Teaching is one of the best ways to understand something deeply.

1.2 — Program Design & Development

Before writing code, a good developer thinks about the structure: what inputs go in, what outputs come out, and what steps happen in between. I used pseudocode and sketches before writing any Swift.

  • Key concept: Program development is iterative. You design, build, test, revise, and repeat. The first version is never the final one.
  • Real-world example: Apps like Instagram go through hundreds of design iterations before any feature ships to users.
  • Reflection: Spending more time on planning saved me time during coding. When I skipped planning, I had to rewrite entire sections.

1.3 — Program Function & Purpose

Every program should have a clear reason for existing. I defined the goal and audience for each project before I started building, which helped me make better design decisions.

  • Key concept: A program's purpose is the problem it solves. Its function is how it solves it. These are different things — you can have the right function but the wrong purpose.
  • Real-world example: TikTok's purpose is entertainment; its function is an infinite scroll algorithm that keeps users watching.
  • Reflection: For Dolandırıcı Radar, the purpose (cybersecurity awareness) shaped every design decision — the question types, the feedback messages, even the color choices.

1.4 — Identifying & Correcting Errors

Bugs are unavoidable. The skill is in finding and fixing them systematically. I learned to use print statements, test edge cases, and isolate problems by commenting out sections of code.

  • Syntax errors: The code won't compile — missing brackets, typos, wrong data types.
  • Logic errors: The code runs but gives wrong output — hardest to catch.
  • Runtime errors: The program crashes during use — often from unexpected input or out-of-bounds access.
  • Reflection: Most of my debugging time goes to logic errors. I now test every edge case before declaring something "done."

02 Big Idea 2 — Data

In this unit I learned how to transform raw data into meaningful information. I practiced reading metadata, cleaning messy datasets, recognizing patterns, and understanding the limits and biases that come with data collection.

2.1 — Binary Numbers & Data Representation

Computers store everything — text, images, sound — as binary (0s and 1s). Understanding this helps you understand why file sizes, color limits, and precision trade-offs exist.

  • Key concept: All digital data is ultimately represented in binary. More bits = more possible values (2ⁿ combinations with n bits).
  • Real-world example: A 24-bit color image stores 8 bits per channel (R, G, B), giving 16.7 million possible colors per pixel.
  • My project: The 4-Bit Binary Counter shows exactly this — 4 bits represent 16 values (0–15).
  • Reflection: Building the binary counter made the abstract concept of bits feel very concrete and visual.

2.2 — Data Compression

Compression reduces file size to save storage and speed up transmission. There are two types: lossless (no data lost) and lossy (some data is permanently removed).

  • Lossless: ZIP, PNG — the original can be perfectly reconstructed.
  • Lossy: JPEG, MP3 — smaller files, but you lose some quality permanently.
  • Key concept: Compression always involves a trade-off between file size, quality, and speed.
  • Real-world example: Streaming platforms like Spotify use lossy compression so songs load fast without being too large.
  • Reflection: I use PNG for icons (lossless, sharp edges) and JPEG for photos (lossy, acceptable quality). Knowing the difference matters.

2.3 — Extracting Information from Data

Raw data doesn't tell you much on its own. You need to organize, filter, and visualize it to find patterns and draw conclusions.

  • Key concept: Patterns and trends only emerge after processing. The same data can tell very different stories depending on how you look at it.
  • Correlation vs causation: Two things changing together doesn't mean one causes the other.
  • Real-world example: A school's test score data might correlate with class size — but that doesn't mean reducing class size always improves scores.
  • Reflection: The hardest part is resisting the urge to jump to conclusions. Good data analysis requires patience and skepticism.

2.4 — Using Programs with Data

Programs can process large datasets far faster than any human. But they can only be as good as the data they're given — bad data in, bad results out.

  • Metadata: Data about data (e.g. photo date, GPS, file size) that helps apps sort and filter without reading the actual content.
  • Data cleaning: Missing, duplicated, or inconsistent values distort analysis. Cleaning is always the first step before drawing any conclusions.
  • Privacy: Hashing, encryption, and anonymization reduce the risk of exposing personal data when sharing datasets.
  • Reflection: My game_infos list from the AP CSP project is a small example of structured data — each entry stores name, date, and players in a consistent format so the program can parse it reliably.

03 Big Idea 3 — Algorithms & Programming

This is the largest Big Idea. It covers the core building blocks of programming: variables, conditions, loops, lists, functions, and algorithms. Most of my Swift projects are direct applications of these concepts.

3.1 — Variables & Assignments

Variables store data that can change during a program's run. In Swift I used @State for variables that should trigger a UI refresh when they change.

  • Key concept: A variable is a named container for a value. Assigning a new value replaces the old one.
  • My project: In the Clock Game, var time = 20 and var playerX: CGFloat = 0 track the game state in real time.
  • Reflection: Choosing good variable names made my code dramatically easier to read and debug.

3.2 — Data Abstraction (Lists)

Instead of tracking individual items separately, lists let you group related data and handle it with loops. This is what makes programs scalable.

  • Key concept: A list stores multiple values under one name. You access them by index, iterate over them with loops, and modify them with append/remove.
  • My project: var game_infos: [String] = [] — one list holds all game records. Without it, I'd need a separate variable for every game.
  • Reflection: Once I understood lists, I started seeing everything as data that could be organized this way — questions in a quiz, flavors in an ice cream app, cities in a clock.

3.3 — Boolean Expressions & Conditionals

Conditionals let programs make decisions. They evaluate a boolean expression (true/false) and branch into different paths accordingly.

  • Key concept: if / else if / else chains control which code runs. Nested conditionals allow more complex logic.
  • My project: The Hot Chocolate Machine uses 4 temperature thresholds — each one leads to a different status message and possibly an auto-stop.
  • Boolean operators: AND (&&), OR (||), NOT (!) let you combine conditions.
  • Reflection: The RPS game's determineWinner function is a clean example of boolean logic — 3 win conditions checked with OR.

3.4 — Iteration (Loops)

Loops repeat a block of code without you having to write it out multiple times. They're what make processing lists possible.

  • Key concept: for loops iterate a known number of times. while loops run until a condition becomes false.
  • My project: The Binary Search protocol uses a while low <= high loop — it keeps halving the search range until the target is found or the range is empty.
  • My project: The Candy Shop simulation runs a for minute in 1...60 loop — simulating 60 minutes of sales in milliseconds.
  • Reflection: Replacing repetitive code with a loop is one of the most satisfying moments in programming.

3.5 — Developing Procedures (Functions)

Procedures (functions) group reusable code under a name. You define it once, then call it whenever you need it — with different inputs each time.

  • Key concept: Functions reduce repetition, make code readable, and allow you to test isolated pieces of logic independently.
  • Parameters: Inputs passed into the function. Return values are the outputs sent back.
  • My project: filter_games(game_infos:, game_name:) is a procedure that takes a list and a name, and returns a filtered list. It can be reused for any game type without rewriting the logic.
  • Reflection: When I find myself copy-pasting the same block of code twice, that's the signal to turn it into a function.

3.6 — Algorithmic Efficiency & Search

Not all algorithms that solve the same problem are equally good. Efficiency matters — especially with large datasets.

  • Linear search: Check every item one by one. Works on any list. Time: O(n).
  • Binary search: Requires a sorted list. Halves the search space each step. Time: O(log n).
  • Example: Finding 23 in a 10-item list — linear search takes up to 10 steps, binary search takes at most 4.
  • My project: The Binary Search Protocol demonstrates this step-by-step with console output showing exactly how many comparisons it takes.
  • Reflection: The difference between O(n) and O(log n) seems small at 10 items, but at 1 billion items it's the difference between 1 billion and 30 steps.

3.7 — Simulations

Simulations model real-world processes in code so we can study them without real-world cost or risk.

  • Key concept: A simulation abstracts reality — it captures the important parts while ignoring irrelevant ones. All simulations have limitations.
  • My project: The Candy Shop Simulation models 60 minutes of a shop — customers arrive at defined intervals, stock depletes, restocking happens on schedule.
  • Limitations: Real customers don't arrive on exact intervals. The simulation is useful for understanding patterns, not predicting exact outcomes.
  • Reflection: Building the simulation made me appreciate how much complexity real-world systems hide — even a simple candy shop has dozens of variables.

04 Big Idea 4 — Computing Systems & Networks

This unit is about how computers are connected and how the internet actually works — from physical hardware to the protocols that let a message travel from Istanbul to Tokyo in milliseconds.

4.1 — The Internet

The internet is a global network of interconnected devices. It was designed to be decentralized — no single point of failure can take the whole thing down.

  • IP Address: A unique identifier for every device on the network (like a postal address).
  • DNS: Translates human-readable domain names (like canortn.github.io) into IP addresses that computers use.
  • HTTP / HTTPS: The protocol that governs how web pages are requested and delivered. HTTPS adds encryption via TLS.
  • Packets: Data is broken into small packets, sent independently across the network, and reassembled at the destination. This is called packet switching.
  • My project: This portfolio site is served from GitHub's servers over HTTPS. Every time you load it, your browser makes HTTP GET requests for the HTML, CSS, fonts, and images.
  • Reflection: I never thought about what happens between typing a URL and seeing a page. Now I understand every step — DNS lookup, TCP handshake, HTTPS negotiation, file transfer, rendering.

4.2 — Fault Tolerance

The internet was designed by ARPA to survive partial failures — even nuclear strikes. This is achieved through redundancy: multiple paths between any two points.

  • Key concept: Redundancy means if one path goes down, packets are automatically rerouted through another. No single point of failure takes the system down.
  • Real-world example: When an undersea cable breaks, internet traffic is rerouted through other cables. Users might experience slowness, but they don't lose connection entirely.
  • Trade-off: More redundancy = more fault tolerance, but also higher cost and more hardware.
  • Reflection: GitHub Pages is itself fault-tolerant — it's served from multiple data centers globally. Even if one goes down, the site stays up.

4.3 — Parallel & Distributed Computing

Some problems are too big or too slow for a single processor. Parallel computing splits a task across multiple processors; distributed computing splits it across multiple machines entirely.

  • Parallel computing: Multiple processors work on different parts of the same problem simultaneously. A quad-core CPU can theoretically run 4 tasks at once.
  • Distributed computing: Tasks spread across many computers over a network. Examples: Google's search index, Bitcoin mining, climate simulations.
  • Key concept: Not all tasks can be parallelized. Some steps depend on previous results — you have to finish step 1 before step 2 starts. This is called sequential dependency.
  • Real-world example: Rendering a 3D movie (like Pixar's) distributes individual frames across thousands of computers in a render farm. Each frame can be processed independently.
  • Reflection: My Swift apps run on a single thread by default. When I added a timer to the Clock Game, I had to think about what runs on the main thread vs. background — a small taste of concurrency.

05 Big Idea 5 — Impact of Computing

Technology doesn't just solve problems — it creates new ones. This unit asks the harder questions: who benefits, who gets left behind, and what responsibilities come with building software.

5.1 — Beneficial & Harmful Effects

Almost every technology has both positive and negative impacts. The same tool that connects people can also spread misinformation.

  • Key concept: Computing innovations are rarely purely good or purely bad. They bring intended benefits and unintended consequences.
  • Example (beneficial): GPS navigation saves fuel, reduces accidents, and helps emergency services respond faster.
  • Example (harmful): GPS also enables mass surveillance and makes people dependent on technology they don't understand.
  • My project: Dolandırıcı Radar is a direct response to the harmful effects of digital scams — using technology to fight a technology-enabled problem.
  • Reflection: Every feature I build, I try to ask: who could misuse this? What happens when it goes wrong?

5.2 — Digital Divide

Not everyone has equal access to technology. The "digital divide" refers to the gap between those who have reliable internet and devices, and those who don't.

  • Key concept: Access divides exist along geographic, economic, and generational lines. Rural areas, lower-income households, and older adults are most affected.
  • Consequence: As more services (banking, education, healthcare) move online, people without access are increasingly excluded.
  • Global example: During COVID-19 remote learning, millions of students worldwide couldn't participate because they lacked a device or stable internet.
  • My project: My Scam Prevention App idea was originally designed for older adults who are frequently targeted by scammers — but are also less likely to be tech-literate. The design had to be extremely simple for exactly this reason.
  • Reflection: Designing for the people least represented in tech is harder — and more important — than designing for people like me.

5.3 — Computing Bias

Algorithms reflect the data and decisions of the people who built them. If that data or those decisions contain bias, the algorithm will too — often invisibly.

  • Key concept: Bias can enter at data collection (who is represented?), model training (what patterns are reinforced?), and deployment (who is affected by errors?).
  • Famous example: Facial recognition systems have been shown to have much higher error rates for darker-skinned faces, because training datasets were predominantly lighter-skinned.
  • Reflection: The scenarios in Dolandırıcı Radar are all based on real Turkish-language scams. If I only included English-language examples, the app would be biased toward a certain demographic and miss the actual threats my target users face.

5.4 — Crowdsourcing

Crowdsourcing uses the collective effort of many people — often online — to complete tasks, gather data, or fund projects.

  • Key concept: The internet makes it possible to coordinate large numbers of people at low cost. This can produce results no single expert or organization could achieve alone.
  • Examples: Wikipedia (collaborative knowledge), reCAPTCHA (solving image recognition tasks while browsing), Folding@home (distributed protein-folding research).
  • Challenge: Quality control is harder. Anyone can contribute, which means errors and misinformation can also spread.
  • Reflection: Stack Overflow is the crowdsourced knowledge base I use almost daily for debugging. It's built entirely on volunteer contributions — which says something powerful about what people will build when given the right platform.

5.5 — Legal & Ethical Concerns

Technology moves faster than the laws designed to govern it. This creates gray areas around privacy, intellectual property, and responsibility.

  • Intellectual property: Copyright, Creative Commons, and open-source licenses govern how software and creative works can be used and shared.
  • Privacy laws: GDPR (Europe) and KVKK (Turkey) regulate how companies collect and store personal data. Violations carry heavy fines.
  • Key concept: Just because you technically can do something with code doesn't mean you legally or ethically should.
  • Reflection: When I built the Dolandırıcı Radar, I made sure not to collect any user data. The app works entirely on-device — no accounts, no analytics, no tracking.

5.6 — Safe Computing

Users and developers both have a responsibility to make computing safer. This means strong passwords, encrypted connections, careful data handling, and awareness of social engineering.

  • Phishing: Fake emails or messages that trick users into giving up credentials. This is the most common attack vector and the direct inspiration for Dolandırıcı Radar.
  • Encryption: Scrambles data so only the intended recipient can read it. HTTPS uses TLS encryption for all web traffic.
  • Strong passwords + 2FA: Two-factor authentication dramatically reduces unauthorized account access even if a password is stolen.
  • Principle of least privilege: Apps should only request the permissions they actually need. An app that asks for microphone access but never uses voice should be treated with suspicion.
  • My project: Dolandırıcı Radar teaches exactly these principles through real examples — recognizing shortened URLs, urgency tactics, fake prizes, and impersonation attempts.
  • Reflection: The more I learned about cybersecurity for this project, the more I realized how many people — including people I know — are vulnerable to basic attacks. Education is the most scalable defense.

func vigenereEncrypt(text:String, key:String) -> String {
    let alphabet = Array("ABCDEFGHIJKLMNOPQRSTUVWXYZ")
    var result = ""
    let keyArray = Array(key.uppercased())
    if keyArray.isEmpty { return text }
    var keyIndex = 0
    for char in text.uppercased() {
        if let ti=alphabet.firstIndex(of:char), let shift=alphabet.firstIndex(of:keyArray[keyIndex]) {
            result.append(alphabet[(ti+shift)%26])
            keyIndex = (keyIndex+1)%keyArray.count
        } else { result.append(char) }
    }
    return result
}
